int~sumaEntera<|int~a,int~b|>|:
    return~a+b!
:|

int~cuadrado<|int~n|>|:
    return~n^2!
:|

float~promedio3<|float~a,float~b,float~c|>|:
    return~<|a+b+c|>/3.0!
:|

bool~esPar<|int~n|>|:
    return~equal<|n%2,0|>!
:|

bool~enRango<|int~valor,int~minimo,int~maximo|>|:
    return~greather_t<|valor,minimo|>@less_te<|valor,maximo|>!
:|

int~factorial<|int~n|>|:
    if~<|less_te<|n,1|>|>|:
        return~1!
    :|else~|:
        return~n*factorial<|n-1|>!
    :|
:|

int~fibonacci<|int~n|>|:
    if~<|less_te<|n,1|>|>|:
        return~n!
    :|else~|:
        return~fibonacci<|n-1|>+fibonacci<|n-2|>!
    :|
:|

int~sumatoria<|int~n|>|:
    if~<|less_te<|n,0|>|>|:
        return~0!
    :|else~|:
        return~n+sumatoria<|n-1|>!
    :|
:|

int~ajustarPorCaso<|int~opcion,int~valor|>|:
    int~resultado!
    resultado<-valor!

    switch~<|opcion|>~|:
        case~1~|:
            resultado<-valor+10!
            break!
        :|
        case~2~|:
            resultado<-valor*2!
            break!
        :|
        case~3~|:
            resultado<-valor-5!
            break!
        :|
        default~|:
            resultado<-valor+1!
        :|
    :|

    return~resultado!
:|
